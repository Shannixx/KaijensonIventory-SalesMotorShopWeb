          SELECT [p].[PurchaseOrderId], [p].[CreatedBy], [p].[CreatedDate], [p].[ExpectedDeliveryDate], [p].[OrderDate], [p].[PurchaseOrderNumber], [p].[Remarks], [p].[Status], [p].[SupplierId], [p].[TotalAmount], [p].[UpdatedDate]
          FROM [PurchaseOrders] AS [p]
          ORDER BY [p].[CreatedDate] DESC
          OFFSET @__p_0 ROWS FETCH NEXT @__p_1 ROWS ONLY
      ) AS [t]
      INNER JOIN [Suppliers] AS [s] ON [t].[SupplierId] = [s].[SupplierId]
      INNER JOIN [Staff] AS [s0] ON [t].[CreatedBy] = [s0].[StaffId]
      ORDER BY [t].[CreatedDate] DESC
